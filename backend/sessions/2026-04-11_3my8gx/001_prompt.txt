  const now = new Date();

  const todayDate = toDate(todayISO);

  const completedCount = tasks.filter((task) => task.completed).length;
  const progressValue = tasks.length ? Math.round((completedCount / tasks.length) * 100) : 0;

  const quickTasks = useMemo(() => {
    return orderedIds
      .map((id) => tasks.find((task) => task.id === id))
      .filter((task): task is Task => Boolean(task));
  }, [orderedIds, tasks]);

  const filteredQuickTasks = useMemo(() => {

        return task.dueDate === todayISO;
      }
      if (filter === "overdue") {
        return toDate(task.dueDate) < todayDate && !task.completed;
      }
      return true;
    });
  }, [quickTasks, filter, todayISO, todayDate]);

  const handleNoteSubmit = (event: React.FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    if (note.trim().length < 5) {
      setNoteError("Введите не менее 5 символов, чтобы заметка была полезной.");
      setNoteSaved(false);
      return;
    }
    setNoteError("");
    setNote("");
    setNoteSaved(true);
    window.setTimeout(() => setNoteSaved(false), 1800);
  };

  const toggleTaskComplete = (taskId: string, next: boolean) => {
    setTasks((prev) =>
      prev.map((task) => {
        if (task.id !== taskId) {
          return task;
        }
        const updatedSubtasks = task.subtasks.map((subtask) => ({
          ...subtask,
          done: next ? true : subtask.done,
        }));
        return {
          ...task,
          completed: next,
          status: next ? "done" : task.status === "done" ? "inProgress" : task.status,
          subtasks: updatedSubtasks,
        };
      })
    );
  };

  const toggleSubtask = (taskId: string, subtaskId: string, next: boolean) => {
    setTasks((prev) =>
      prev.map((task) => {
        if (task.id !== taskId) {
          return task;
        }
        const updatedSubtasks = task.subtasks.map((subtask) =>
          subtask.id === subtaskId ? { ...subtask, done: next } : subtask
        );
        const allDone = updatedSubtasks.every((subtask) => subtask.done);
        return {
          ...task,
          subtasks: updatedSubtasks,
          completed: allDone,
          status: allDone ? "done" : task.status === "done" ? "inProgress" : task.status,
        };
      })
    );

      if (targetIndex === -1) {
        nextOrder.push(sourceId);
        return [...nextOrder];
      }
      nextOrder.splice(targetIndex, 0, sourceId);
      return [...nextOrder];
    });

([developer.android.com](https://developer.android.com/developer-verification?utm_source=openai))

═══ TYPOGRAPHY ═══
Heading Font: Inter
Body Font: Inter


═══ STYLE GUIDE ═══
DESIGN LANGUAGE: Colorful Vibrant
- Use bold gradient backgrounds: linear-gradient with category primary + secondary
- Surface: white cards on colored background, or semi-transparent
- Accent: vibrant complementary colors, not just one color
- Typography: Inter 700/800 for headings, round feel
- Border radius: 16-20px cards, pill buttons (9999px), 12px inputs
- Shadows: colored shadows matching primary (box-shadow 0 8px 25px primary at 30%)
- Gradients: use on buttons, heroes, badges, progress bars
- Icons: slightly larger 20-24px, colorful or white on colored bg
- Animations: bouncy micro-interactions (spring easing)
- Use emoji or colorful illustrations as accents
- Categories: color-code different sections
- Buttons: gradient fill, white text, slight shadow
- Cards: white with colored left border or gradient header strip
- High contrast ratios, accessible despite vibrancy

═══ TECHNICAL REQUIREMENTS ═══
- React functional components with TypeScript
- Tailwind CSS for all styling
- Import icons from lucide-react
- Use inline styles ONLY for dynamic values (colors from palette above)
- No external image URLs — use gradient placeholders or emoji
- Component must be self-contained, no external dependencies beyond React/Tailwind/lucide-react
- Export default the main component
- Include realistic sample data (5-10 items minimum)
- Responsive: works on mobile (375px) and desktop (1280px)
